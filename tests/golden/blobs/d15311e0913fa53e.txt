import marimo

app = marimo.App()

with app.setup:
    import mogs


@app.function
def double(x):
    ...


@app.cell
def _():
    mogs.grade("q1", points=1, title="Question 1", test_file="tests/q1.py", uses=[], hidden=1)
    return


@app.cell(hide_code=True)
def _():
    import marimo as mo

    return (mo,)


@app.cell(hide_code=True)
def _(mo):
    mo.md(r"""
    ## Submit

    Save the notebook, then click **Export submission** below. It runs the public tests on
    the saved file and makes a zip of your notebook. Upload that zip to Gradescope.

    No button below? Run the notebook first: **Run all stale cells**, or Ctrl+Shift+R
    (Cmd+Shift+R on a Mac).
    """)
    return


@app.cell(hide_code=True)
def _():
    export_button = mogs.export_button()
    export_button
    return (export_button,)


@app.cell(hide_code=True)
def _(export_button):
    mogs.export(export_button, name="hw", manual=[], titles={"q1": "Question 1"}, placeholders={})
    return


if __name__ == "__main__":
    app.run()
